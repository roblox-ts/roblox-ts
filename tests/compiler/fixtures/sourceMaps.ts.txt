export function named(flag: boolean) {
    print("probe:named");
    if (flag) {
        print("probe:if");
    } else if (!flag) {
        print("probe:elseif");
    } else {
        print("probe:else");
    }
    for (const item of [1, 2]) {
        print("probe:for", item);
    }
    while (flag) {
        print("probe:while");
        break;
    }
    do {
        print("probe:repeat");
    } while (flag);
}
